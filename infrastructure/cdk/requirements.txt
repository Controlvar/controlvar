# The CDK app's own pins.
#
# aws-cdk-lib and constructs are ALSO pinned in backend/requirements.txt, which
# is what CI and the venv install -- this file is what a standalone
# `pip install -r infrastructure/cdk/requirements.txt` gets. The two must agree
# or `cdk synth` behaves differently in the two environments: this file sat at
# 2.94.0 while everything that actually ran used 2.268.0, and API differences
# between them are exactly how a stack that never synthesises goes unnoticed.
# Dependabot updates both (see .github/dependabot.yml).
#
# Both are `==` on purpose. test_requirement_pins_agree.py can only compare
# exact pins, so a RANGE here is invisible to it: `constructs>=10.8.1,<11.0.0`
# sat in this file while backend/requirements.txt pinned no constructs at all
# and let pip resolve it transitively -- the venv landed on 10.5.1, below this
# file's own floor, and the gate said nothing. That is the drift this file
# exists to prevent, reappearing in the one package the gate could not see.
aws-cdk-lib==2.271.0
constructs==10.8.1           # aws-cdk-lib 2.270.0 declares >=10.5.0,<11.0.0
pytest==9.1.1
